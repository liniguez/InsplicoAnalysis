1 Material

Insplico uses a tab delimited file where all exons are listed and which are going to be quantified. Here you can get a subset of one used to run insplico. Insplico is ran for each sample individually and then the output of all the runs can be merged into a single big data. Here you can get a subset of a run. Here I analyzed two single cell libraries NB292 and NB293. I got the clustering of the cells based on gene expression. Alignment file was then split per sample into different clusters, based on cell barcodes and used as input for Insplico.

Note

It is important to mention that each row form Insplico output correspond to each row from tab delimited exon file. More info about insplico in here

3 Filter Exons

We then need to get the count of supporting reads for inclusion/exclusion and for that we’ll do; for long reads:

(EXON_INCLSPLICED_2UPSTRM+EXON_INCLSPLICED_2DOSTRM)/2 + EXON_EXCL

for short reads it would be something like:

EXON_INCLSPLICED_2UPSTRM +EXON_INCLSPLICED_2DOSTRM + EXON_EXCL

Here we are using the mentioned values for assigning a quality score to each PSI based on vast-tools quality parameters (more info).

Code 
covScore <- function(total_reads, exon_excl, exon_inclspliced_2upstrm, exon_inclspliced_2dostrm){
  case_when(
    total_reads >= 100 & (exon_excl >= 20 | ( (exon_inclspliced_2upstrm >= 20 & exon_inclspliced_2dostrm >= 15) | (exon_inclspliced_2upstrm >= 15 & exon_inclspliced_2dostrm >= 20) )) ~ "SOK",
    total_reads < 100 & (exon_excl >= 20 | ( (exon_inclspliced_2upstrm >= 20 & exon_inclspliced_2dostrm >= 15) | (exon_inclspliced_2upstrm >= 15 & exon_inclspliced_2dostrm >= 20) )) ~ "OK",
    exon_excl >= 15 | ( (exon_inclspliced_2upstrm >= 15 & exon_inclspliced_2dostrm >= 10) | (exon_inclspliced_2upstrm >= 10 & exon_inclspliced_2dostrm >= 15) ) ~ "LOW",
    exon_excl >= 10 | ( (exon_inclspliced_2upstrm >= 10 & exon_inclspliced_2dostrm >= 5) | (exon_inclspliced_2upstrm >= 5 & exon_inclspliced_2dostrm >= 10) ) ~ "VLOW",
    TRUE ~ "N"
  )
}

rawinsp<-insp %>% filter(!is.na(feature)) %>%

For this part I used similar methods as the ones reported in the paper of betAS. This methods are still under discussion and comments for improvement are more than welcome. The needed table can be found in here.

Code 
betasmat<-read.table("table_betAS.txt") #This table is provided by the betAS package.
givenCovfindIncr <- function(cov, maxDevRef){
  if(cov > max(maxDevRef[,1])){
    incr <- 1
  }else{
    p <- which(cov == maxDevRef$cov)
    incr <- maxDevRef$maxIncr[p]
  }
  return(incr)
}

#I applied a recursive function with a limit of 10^7 sim for a pseudo p-value aproximation.
getpseudop<-function(inc1,excl1,inc2,excl2, seed=12032024){
  nsim=1000
  p_zero=1
  set.seed(seed)
  while(p_zero==1 & nsim<10^7){ 
    r1<-rbeta(nsim,inc1,excl1)
    r2<-rbeta(nsim,inc2,excl2)
    p_zero<-sum((r1-r2>0))/nsim
    nsim<-nsim*10
  }
  psudopval<-1-abs((p_zero-0.5)*2)
}

We need again the number of reads per sample that support inclusion/exclusion of the exon. We’ll need to modify a bit betAS pipeline. Pseudo-pvalue will be adjusted with fdr.

Code 
getINC<-function(x){
  y<-gsub(pattern="@",replacement = ",",x) %>% strsplit(split=",") %>% unlist()
  inc<-as.numeric(y[3])+as.numeric(y[4])
  return(inc)
}
getEXC<-function(x){
  y<-gsub(pattern="@",replacement = ",",x) %>% strsplit(split=",") %>% unlist()
  exc<-as.numeric(y[5])
  return(exc)
}

resSimBeta_temp<-rawinsp %>% filter(cluster==1 & ROW %in% psiFilt_noCryp_noConst$ROW)%>%
  rowwise()%>%
  mutate(INC=getINC(Q),
         EXCL=getEXC(Q)*2,  # EXON_EXCL*2
         INC=ifelse(INC==0,givenCovfindIncr(round(EXCL),betasmat),INC),
         EXCL=ifelse(EXCL==0,givenCovfindIncr(round(INC),betasmat),EXCL)) %>%
  ungroup() %>%
  select(ROW,sample,cluster,INC,EXCL) %>%
  pivot_wider(id_cols = c("ROW","cluster"),names_from = "sample",values_from = c("INC","EXCL"))

resBETAS<-sapply(1:length(resSimBeta_temp$ROW),function(x){
  res<-getpseudop(inc1=as.numeric(resSimBeta_temp[x,"INC_NB292"]),
                  excl1=as.numeric(resSimBeta_temp[x,"EXCL_NB292"]),
                  inc2=as.numeric(resSimBeta_temp[x,"INC_NB293"]),
                  excl2=as.numeric(resSimBeta_temp[x,"EXCL_NB293"]))
  return(res)
})
resSimBeta_temp$psudopval<-resBETAS
resSimBeta<-resSimBeta_temp %>% group_by(cluster) %>% mutate(sudopadj= p.adjust(psudopval,method = "fdr"))

5 With Replicates

Now lets assume we have replicates and the different clusters are replicates of the sample. Here again we need to filter constitutive and cryptic exons. First, we’ll check what is the proportion of clusters from each sample where the number of reads supporting the inclusion/exclusion is greater than 15, and filter if the proportion is greater than 30%. This is too lax, but it is because of the toy data used, usually this proportion should be above 50%.

For assesing DSA with replicates I used beta regression on the samples and p-values were adjusted with fdr

Code 
ref="NB292"
test="NB293"
Rep_res<-insp_noN %>% select(ROW,sample, PSI) %>%
    right_join(evin2check, by=c("ROW")) %>% 
    filter(sample %in% c(ref,test)) %>%
    split(.$ROW) %>%
    lapply(function(x){
      evrow=x$ROW %>% unique()
      temp<-x%>%
        mutate(TestFactor=factor(sample,levels=c(ref,test))) %>%
        filter(!is.na(TestFactor))
      if(length(unique(temp$TestFactor))>1){
        pvalor<-temp %>%
          mutate(PSI_EX=PSI/100,PSI_EX=ifelse(PSI_EX==0,0.01,
                                              ifelse(PSI_EX==1,0.99,PSI_EX))) %>%
          betareg(PSI_EX ~TestFactor|1,data = .) %>% summary() %>%
          .$coefficients %>% .$mean
        pvalor<-pvalor[paste0("TestFactor",test),'Pr(>|z|)']
      }else{pvalor <-1}
      means<-temp %>% group_by(TestFactor) %>% reframe(M=mean(PSI))
      ref_means<-means %>% filter(TestFactor==ref)
      test_means<-means  %>% filter(TestFactor!=ref)
      dPSI<-test_means %>%
        mutate(dpsi=M-ref_means$M, name=paste0("TestFactor",TestFactor)) %>%
        select(name,dpsi)
      result_out<-tibble("ROW"=unique(x$ROW),test_PSI=test_means$M,
                         ref_PSI=ref_means$M, "dPSI"=dPSI$dpsi,"pvalue"=pvalor)
